Audit Logs

Record real events from a browser form and immediately inspect recent, actor-specific and subject-specific audit records.

Install

composer require tihloh/prefab-logs

Create public/logs.php

<?php
require __DIR__ . '/../vendor/autoload.php';

use Tihloh\Prefab\Logs\Repositories\PdoLogRepository;
use Tihloh\Prefab\Logs\Services\LogManager;

$pdo = new PDO(
    'mysql:host=127.0.0.1;dbname=prefab_tutorial;charset=utf8mb4',
    'root', '', [PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION]
);

$logs = new LogManager(new PdoLogRepository($pdo));

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $logs->record([
        'action' => $_POST['action'] ?? 'document.updated',
        'subject_type' => 'document',
        'subject_id' => $_POST['subject_id'] ?? '1001',
        'actor_id' => $_POST['actor_id'] ?? '1',
        'message' => $_POST['message'] ?? 'Tutorial event',
        'changes' => [
            'status' => ['old' => 'pending', 'new' => 'approved'],
        ],
        'metadata' => ['source' => 'tutorial-form'],
        'ip_address' => $_SERVER['REMOTE_ADDR'] ?? null,
        'user_agent' => $_SERVER['HTTP_USER_AGENT'] ?? null,
    ]);
}

$recent = $logs->recent(20);
?>
<!doctype html><html><body>
<h1>Prefab Logs Test</h1>
<form method="post">
    <p><input name="action" value="document.approved"></p>
    <p><input name="actor_id" value="1" placeholder="Actor ID"></p>
    <p><input name="subject_id" value="1001" placeholder="Subject ID"></p>
    <p><input name="message" value="Document was approved." size="50"></p>
    <button>Record event</button>
</form>
<h2>Recent logs</h2>
<pre><?= htmlspecialchars(print_r($recent, true)) ?></pre>
</body></html>

PdoLogRepository automatically creates the prefab_logs storage table for supported databases.

Query examples

$logs->recent();
$logs->recent(50);
$logs->find(1);
$logs->forSubject('document', 1001);
$logs->forActor(1);

Structured changes and metadata

$logs->record([
    'action' => 'user.updated',
    'subject_type' => 'user',
    'subject_id' => 25,
    'actor_id' => 7,
    'changes' => [
        'office' => ['old' => 'Accounting', 'new' => 'Budget'],
    ],
    'metadata' => [
        'source' => 'admin-ui',
        'request_id' => 'abc123',
    ],
]);

Human-friendly view

$human = $logs->humanRecent(
    50,
    actorResolver: fn ($id) => "User #{$id}",
    subjectResolver: fn ($type, $id) => "{$type} #{$id}",
);

Diagnostics

print_r($logs->explain());
